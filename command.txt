// Топ-20 рекомендаций новостей для пользователя с id = 1
MATCH (target:User {id: 1})

// 1. Собираем все положительные взаимодействия пользователя
OPTIONAL MATCH (target)-[:LIKED]->(likedArticles:Article)
OPTIONAL MATCH (target)-[:FAVORITED]->(favoritedArticles:Article)
OPTIONAL MATCH (target)-[:COMMENTED]->(commentedArticles:Article)
WITH target, 
     collect(DISTINCT likedArticles) + 
     collect(DISTINCT favoritedArticles) + 
     collect(DISTINCT commentedArticles) AS positiveArticles

// 2. Получаем теги, которые нравятся пользователю
OPTIONAL MATCH (positive:Article)-[:TAGGED_WITH]->(preferredTag:Tag)
WHERE positive IN positiveArticles
WITH target, positiveArticles, collect(DISTINCT preferredTag) AS preferredTags

// 3. Получаем категории, на которые подписан пользователь
OPTIONAL MATCH (target)-[:SUBSCRIBED_TO]->(preferredCategory:Category)
WITH target, positiveArticles, preferredTags, collect(DISTINCT preferredCategory) AS preferredCategories

// 4. Получаем историю чтения пользователя
OPTIONAL MATCH (target)-[:READ]->(readArticle:Article)
WITH target, positiveArticles, preferredTags, preferredCategories, 
     collect(DISTINCT readArticle) AS readHistory

// 5. Находим похожих пользователей (за 30 дней)
OPTIONAL MATCH (target)-[:READ]->(common:Article)<-[:READ]-(similarUser:User)
WHERE similarUser.id <> target.id
  AND common.created_at >= datetime() - duration({days: 30})
WITH target, positiveArticles, preferredTags, preferredCategories, readHistory,
     collect(DISTINCT similarUser) AS similarUsers

// 6. Получаем статьи от похожих пользователей
OPTIONAL MATCH (similar:User)-[:READ]->(collabArticle:Article)
WHERE similar IN similarUsers
  AND collabArticle.is_published = true
  AND collabArticle.created_at >= datetime() - duration({days: 7})
  AND NOT collabArticle IN readHistory
  AND NOT collabArticle IN positiveArticles
WITH target, preferredTags, preferredCategories, 
     collect(DISTINCT collabArticle) AS collaborativeArticles

// 7. Основной сбор кандидатов - ТОЛЬКО за последние 7 дней
MATCH (candidate:Article)
WHERE candidate.is_published = true
  AND candidate.created_at >= datetime() - duration({days: 7})
  AND NOT (target)-[:READ]->(candidate)
  AND NOT (target)-[:LIKED]->(candidate)
  AND NOT (target)-[:FAVORITED]->(candidate)

// 8. Получаем теги и категории кандидата
OPTIONAL MATCH (candidate)-[:TAGGED_WITH]->(tag:Tag)
OPTIONAL MATCH (candidate)-[:IN_CATEGORY]->(category:Category)

WITH candidate,
     collect(DISTINCT tag) AS candidateTags,
     collect(DISTINCT category) AS candidateCategories,
     preferredTags, preferredCategories, collaborativeArticles

// 9. ВЫЧИСЛЯЕМ СКОР
WITH candidate,
     REDUCE(score = 0, t IN preferredTags | 
            CASE WHEN t IN candidateTags THEN score + 5 ELSE score END) AS tagScore,
     
     REDUCE(score = 0, c IN preferredCategories | 
            CASE WHEN c IN candidateCategories THEN score + 4 ELSE score END) AS categoryScore,
     
     CASE WHEN candidate IN collaborativeArticles THEN 10 ELSE 0 END AS collaborativeScore,
     
     (coalesce(candidate.views, 0) / 2000.0) * 0.3 + 
     (coalesce(candidate.likes, 0) / 200.0) * 0.5 AS popularityScore

// 10. Финальный скор
WITH candidate, 
     (tagScore + categoryScore + collaborativeScore + popularityScore) AS finalScore,
     tagScore, categoryScore, collaborativeScore, popularityScore

WHERE finalScore > 0

// 11. Возвращаем топ-20
RETURN 
    candidate.id AS article_id,
    candidate.title AS title,
    candidate.created_at AS published_date,
    candidate.views AS views,
    candidate.likes AS likes,
    ROUND(finalScore, 2) AS total_score,
    ROUND(tagScore, 2) AS liked_tags_score,
    ROUND(categoryScore, 2) AS categories_score,
    ROUND(collaborativeScore, 2) AS similar_users_score,
    ROUND(popularityScore, 2) AS popularity_score,
    CASE 
        WHEN tagScore > 0 AND collaborativeScore > 0 AND categoryScore > 0 
            THEN '✅ Все условия'
        WHEN tagScore > 0 AND collaborativeScore > 0 
            THEN '✅ Любимые теги + Похожие пользователи'
        WHEN tagScore > 0 AND categoryScore > 0 
            THEN '✅ Любимые теги + Подписки'
        WHEN collaborativeScore > 0 AND categoryScore > 0 
            THEN '✅ Похожие пользователи + Подписки'
        WHEN tagScore > 0 
            THEN '✅ Только любимые теги'
        WHEN collaborativeScore > 0 
            THEN '✅ Только похожие пользователи'
        WHEN categoryScore > 0 
            THEN '✅ Только подписки'
        ELSE '📌 Без совпадений'
    END AS matching_conditions
ORDER BY finalScore DESC
LIMIT 20
